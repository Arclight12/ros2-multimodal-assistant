# Runtime Python dependencies for the robot (target: Raspberry Pi 4/5, Ubuntu
# 24.04 64-bit, ROS 2 Jazzy). ROS packages such as rclpy and cv_bridge come from
# rosdep/apt, not pip. Training-only packages are in requirements-train.txt.

# --- Perception (ArUco workspace mapping, YOLO detection) ---
numpy
opencv-contrib-python<5
# NOTE: `opencv-python` is intentionally NOT listed. `opencv-contrib-python`
# is a superset that also provides the ArUco modules; installing both wheels
# is redundant and can conflict. Kept below 5 to match ROS's apt cv_bridge
# (built against OpenCV 4).

# --- Interaction: gaze (MediaPipe FaceMesh) ---
# Below 1.0: MediaPipe 1.x removed the `mp.solutions` API the gaze node uses.
# A range, not a pin: aarch64 (Raspberry Pi) wheels stop at 0.10.18.
mediapipe>=0.10.13,<0.11

# --- Perception (YOLO26n via Ultralytics) ---
# Ultralytics depends on torch, so torch is installed here as a transitive dependency.
ultralytics

# --- Motion: serial link to the Arduino arm controller ---
pyserial

# --- Interaction: voice (NVIDIA hosted speech NIM over gRPC) ---
# Pinned: riva >= 2.17 needs protobuf 5/6, which breaks mediapipe 0.10.x (protobuf < 5).
nvidia-riva-client==2.16.0
protobuf==4.25.9
grpcio-tools==1.62.3

# --- YAML configuration parsing ---
pyyaml
