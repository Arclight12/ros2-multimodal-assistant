# Python dependencies for the multimodal robotic assistant.

# ships with ROS 2, plus the workspace packages themselves.

numpy
opencv-contrib-python

mediapipe

# --- Interaction: voice (future: Vosk offline speech recognition) ---
vosk

ultralytics

pyserial

# --- YAML configuration parsing ---
pyyaml

# --- Training only (not imported by ROS runtime nodes) ---
torch
